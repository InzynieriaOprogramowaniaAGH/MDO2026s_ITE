    matchLabels:
      app: realworld
  template:
    metadata:

      containers:
      - name: realworld-container
        image: realworld-app:latest
        imagePullPolicy: Never # Zla wykorzystywania lokalnego obrazu

  ports:

      targetPort: 8080
  type: ClusterIP
